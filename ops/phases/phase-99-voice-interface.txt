## Step 2 — Voice hook (src/dashboard/hooks/useVoice.ts)
```typescript
interface UseVoiceOptions {
  onTranscript: (text: string) => void;  // called with final transcript
  onInterim?: (text: string) => void;    // called with interim result
  lang?: string;                          // default 'en-US'
}

export function useVoice(options: UseVoiceOptions) {
  // Returns:
  return {
    isListening: boolean,
    isSupported: boolean,      // false if browser lacks SpeechRecognition
    startListening: () => void,
    stopListening: () => void,
    error: string | null,
  }
}
```
- Use `window.SpeechRecognition || window.webkitSpeechRecognition`
- Set `continuous: false`, `interimResults: true`
- Auto-stop on silence (onend event)
- Gracefully degrade: if not supported, isSupported=false, show "Voice not available in this browser"

## Step 3 — TTS hook (src/dashboard/hooks/useTTS.ts)
```typescript
export function useTTS() {
  return {
    speak: (text: string, options?: { rate?: number; pitch?: number }) => void,
    stop: () => void,
    isSpeaking: boolean,
    isSupported: boolean,
  }
}
```
- Use `window.speechSynthesis`
- Default voice: first English voice available
- Strip markdown before speaking (remove **, *, #, backticks, links)
- Cancel current speech before starting new one
